//! Reads INI-style settings files: `key = value` pairs under optional `[section]` headers.

use std::collections::BTreeMap;
use std::fmt;
use std::str::FromStr;

#[derive(Debug, PartialEq, Eq)]
pub struct ParseError {
    pub line: usize,
    pub kind: ParseErrorKind,
}

#[derive(Debug, PartialEq, Eq)]
pub enum ParseErrorKind {
    MissingEquals,
    EmptyKey,
    Duplicate(String),
    BadSection,
}

impl fmt::Display for ParseError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match &self.kind {
            ParseErrorKind::MissingEquals => write!(f, "line {}: expected `=`", self.line),
            ParseErrorKind::EmptyKey => write!(f, "line {}: empty key", self.line),
            ParseErrorKind::Duplicate(key) => {
                write!(f, "line {}: `{key}` set twice", self.line)
            }
            ParseErrorKind::BadSection => {
                write!(f, "line {}: malformed section header", self.line)
            }
        }
    }
}

impl std::error::Error for ParseError {}

#[derive(Debug, Default, PartialEq, Eq)]
pub struct Config {
    values: BTreeMap<String, String>,
}

impl Config {
    /// Looks up a key. A key under a section is written `section.key`.
    pub fn get(&self, key: &str) -> Option<&str> {
        self.values.get(key).map(String::as_str)
    }

    /// Looks up a key, or gives `default` when the key is unset.
    pub fn get_or<'a>(&'a self, key: &str, default: &'a str) -> &'a str {
        self.get(key).unwrap_or(default)
    }

    pub fn parse_as<T: FromStr>(&self, key: &str) -> Option<T> {
        self.get(key)?.parse().ok()
    }

    pub fn len(&self) -> usize {
        self.values.len()
    }

    pub fn is_empty(&self) -> bool {
        self.values.is_empty()
    }

    /// The keys and values of one section, with the section prefix removed.
    pub fn section<'a>(&'a self, name: &str) -> impl Iterator<Item = (&'a str, &'a str)> {
        let prefix = format!("{name}.");
        self.values
            .iter()
            .filter_map(move |(k, v)| Some((k.strip_prefix(prefix.as_str())?, v.as_str())))
    }
}

impl FromStr for Config {
    type Err = ParseError;

    fn from_str(text: &str) -> Result<Self, Self::Err> {
        let mut values = BTreeMap::new();
        let mut section = String::new();
        for (index, raw) in text.lines().enumerate() {
            let line = raw.trim();
            if line.is_empty() || line.starts_with(['#', ';']) {
                continue;
            }
            let error = |kind| ParseError {
                line: index + 1,
                kind,
            };
            if let Some(header) = line.strip_prefix('[') {
                let name = header
                    .strip_suffix(']')
                    .ok_or_else(|| error(ParseErrorKind::BadSection))?;
                section = name.trim().to_owned();
                continue;
            }
            let (key, value) = line
                .split_once('=')
                .ok_or_else(|| error(ParseErrorKind::MissingEquals))?;
            let key = key.trim();
            if key.is_empty() {
                return Err(error(ParseErrorKind::EmptyKey));
            }
            let key = if section.is_empty() {
                key.to_owned()
            } else {
                format!("{section}.{key}")
            };
            let value = value.trim().to_owned();
            if values.insert(key.clone(), value).is_some() {
                return Err(error(ParseErrorKind::Duplicate(key)));
            }
        }
        Ok(Self { values })
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn reads_pairs_and_skips_comments() {
        let config: Config = "# server\n; legacy\nhost = example.org\nport=8080\n"
            .parse()
            .unwrap();
        assert_eq!(config.get("host"), Some("example.org"));
        assert_eq!(config.parse_as::<u16>("port"), Some(8080));
        assert_eq!(config.len(), 2);
    }

    #[test]
    fn missing_equals_names_the_line() {
        let err = "a = 1\nbroken\n".parse::<Config>().unwrap_err();
        assert_eq!(
            err,
            ParseError {
                line: 2,
                kind: ParseErrorKind::MissingEquals
            }
        );
    }

    #[test]
    fn duplicate_key_is_an_error() {
        let err = "a = 1\na = 2\n".parse::<Config>().unwrap_err();
        assert_eq!(err.kind, ParseErrorKind::Duplicate("a".to_owned()));
    }

    #[test]
    fn sections_prefix_their_keys() {
        let config: Config = "top = 1\n[db]\nhost = local\n[ cache ]\nhost = mem\n"
            .parse()
            .unwrap();
        assert_eq!(config.get("db.host"), Some("local"));
        assert_eq!(
            config.section("cache").collect::<Vec<_>>(),
            [("host", "mem")]
        );
        let err = "[db\n".parse::<Config>().unwrap_err();
        assert_eq!(err.kind, ParseErrorKind::BadSection);
    }
}
